% Bibliography. Biber is configured in latexmkrc.
% Author-year (APA) citation style, as required by the chair's guide. URLs are
% suppressed so the reference list stays clean; DOIs are retained.
\usepackage[
  backend=biber,
  style=apa,
  sorting=nyt,
  sortcites=true,
  doi=true,
  url=false,
  isbn=false,
  eprint=false
]{biblatex}
\DeclareLanguageMapping{english}{english-apa}

% Single-spaced reference entries with a clear gap between them.
\setlength{\bibitemsep}{0.65\baselineskip}
\setlength{\bibhang}{1.2em}
\renewcommand*{\bibfont}{\normalfont\setstretch{1.0}\selectfont}

% Hyperlinks should load late; cleveref should load after hyperref.
\usepackage{hyperref}
\usepackage[nameinlink,noabbrev]{cleveref}
