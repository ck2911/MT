% Figure import helper.
% Call site: \thesisfigure[width]{subdir/name.png}{caption}{label}.
% Resolution order per call: vector PDF sibling of the given PNG, then the PNG
% itself, then the missing-artifact placeholder box. The default width is a
% single-panel evidence size; wider families (comparison charts, matrices)
% pass an explicit [width] at the call site as before.
% Chair guide: the caption is the heading above the figure and the source is
% stated directly below it.
\newcommand{\figuresource}[2][\linewidth]{%
  \par\vspace{0.3em}%
  \parbox{#1}{\footnotesize\setstretch{1.0}Source: #2}\par
}
\newcommand{\thesisfigure}[4][0.80\textwidth]{%
  \begin{figure}[htbp]
    \centering
    \caption{#3}
    \label{#4}
    \StrSubstitute{#2}{.png}{.pdf}[\thesisfigurepdfname]%
    \IfFileExists{\figpath/\thesisfigurepdfname}{%
      \includegraphics[width=#1]{\figpath/\thesisfigurepdfname}%
    }{%
      \IfFileExists{\figpath/#2}{%
        \includegraphics[width=#1]{\figpath/#2}%
      }{%
        \fbox{\begin{minipage}[c][0.25\textheight][c]{0.86\textwidth}
          \centering
          Missing canonical figure artifact:\par
          \nolinkurl{results/final/figures/#2}
        \end{minipage}}%
      }%
    }%
    \figuresource[#1]{own calculations.}
  \end{figure}
}

% Table artifact helper. This keeps CSV files as the source of truth while
% registering the thesis-facing table in the list of tables.
\newcommand{\artifacttable}[4]{%
  \begin{table}[htbp]
    \centering
    \caption{#1}
    \label{#2}
    \begin{tabularx}{\textwidth}{@{}p{0.28\textwidth}X@{}}
      \toprule
      Canonical artifact & \nolinkurl{#3} \\
      Thesis use & #4 \\
      \bottomrule
    \end{tabularx}
    \vspace{0.25em}
    {\footnotesize Source: \nolinkurl{#3}.}
  \end{table}
}
